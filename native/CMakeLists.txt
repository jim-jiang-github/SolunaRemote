cmake_minimum_required(VERSION 3.24)
project(soluna-stream VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
option(SOLUNA_BUILD_EMBEDDED "Build the native client as a library" OFF)
include(FetchContent)
FetchContent_Declare(moonlight
  GIT_REPOSITORY https://github.com/moonlight-stream/moonlight-common-c.git
  GIT_TAG d5b38a3ac63927a11bebd8337df024f11eba34d5
  GIT_SHALLOW FALSE
  GIT_SUBMODULES_RECURSE TRUE)
FetchContent_MakeAvailable(moonlight)
find_package(SDL2 CONFIG REQUIRED)
if(APPLE OR SOLUNA_BUILD_EMBEDDED)
  find_package(PkgConfig REQUIRED)
  if(SOLUNA_BUILD_EMBEDDED)
    pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
  else()
    pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET
      libavcodec libavformat libavutil libswscale libavdevice libswresample libavfilter)
  endif()
  pkg_check_modules(OPUS REQUIRED IMPORTED_TARGET opus)
  list(GET OPUS_INCLUDE_DIRS 0 OPUS_INCLUDE_DIR)
  get_filename_component(OPUS_INCLUDE_PARENT "${OPUS_INCLUDE_DIR}" DIRECTORY)
  find_path(OPUS_HEADER_ROOT opus/opus_multistream.h
    HINTS "${OPUS_INCLUDE_DIR}" "${OPUS_INCLUDE_PARENT}" REQUIRED)
  set(FFMPEG_TARGET PkgConfig::FFMPEG)
  set(OPUS_TARGET PkgConfig::OPUS)
else()
  find_package(FFMPEG REQUIRED)
  find_package(Opus CONFIG REQUIRED)
  set(FFMPEG_TARGET ${FFMPEG_LIBRARIES})
  set(OPUS_TARGET Opus::opus)
endif()
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Threads REQUIRED)
if(SOLUNA_BUILD_EMBEDDED)
  if(NOT WIN32)
    message(FATAL_ERROR "The embedded native client currently supports Windows only.")
  endif()
  add_library(soluna-stream SHARED src/main.cpp)
  set_target_properties(soluna-stream PROPERTIES PREFIX "" OUTPUT_NAME "soluna-stream")
  target_compile_definitions(soluna-stream PRIVATE SOLUNA_STREAM_EMBEDDED)
  target_link_options(soluna-stream PRIVATE -static-libgcc -static-libstdc++ -Wl,-s)
else()
  add_executable(soluna-stream src/main.cpp)
endif()
target_include_directories(soluna-stream PRIVATE ${FFMPEG_INCLUDE_DIRS})
if(APPLE OR SOLUNA_BUILD_EMBEDDED)
  target_include_directories(soluna-stream PRIVATE "${OPUS_HEADER_ROOT}")
endif()
target_compile_definitions(soluna-stream PRIVATE SDL_MAIN_HANDLED NOMINMAX)
target_link_libraries(soluna-stream PRIVATE moonlight-common-c SDL2::SDL2 ${OPUS_TARGET}
  OpenSSL::Crypto nlohmann_json::nlohmann_json Threads::Threads ${FFMPEG_TARGET})
if(WIN32)
  target_link_libraries(soluna-stream PRIVATE comctl32 gdi32 user32)
endif()
if(MSVC)
  target_compile_options(soluna-stream PRIVATE /W4 /utf-8 /permissive-)
  target_link_options(soluna-stream PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup)
else()
  target_compile_options(soluna-stream PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(NOT SOLUNA_BUILD_EMBEDDED)
  install(TARGETS soluna-stream RUNTIME DESTINATION .)
endif()
